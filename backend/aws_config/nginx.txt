# /etc/nginx/sites-available/backend

server {
    listen 80;
    server_name 43.201.89.96;

    location /static/ {
        alias /var/www/backend/static/;
    }

    location /media/ {
        alias /var/www/backend/media/; 
    }

    location / {
        include proxy_params;
        proxy_pass http://unix:run/backend/gunicorn.sock;


        chunked_transfer_encoding on;
        gzip off;
        proxy_buffering off;
        proxy_request_buffering off;
        proxy_cache off;
        proxy_read_timeout 30000;
        proxy_send_timeout 30000;
    }
}
